fn isEven := (i32 n) -> bool
{
    if n == 0
    {
        exit true
    }
    exit isOdd(n - 1)
}
fn isOdd := (i32 n) -> bool
{
    if n == 0
    {
        exit false
    }
    exit isEven(n - 1)
}
fn firstOdd := (i32 n) -> i32
{
    i32 mut i{n}
    while true
    {
        if isOdd(i)
        {
            exit i
        }
        i := i + 1
    }
    exit 0
}

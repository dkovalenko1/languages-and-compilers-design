struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Holder
{
    Point p
    fn work := () -> i32
    {
        Point mut q{p}
        exit q.shift(5)
    }
}
Holder h{{3, 4}}
print(h.work())
exit h.p.x

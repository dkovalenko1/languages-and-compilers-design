struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Inner
{
    Point mut p
}
struct Outer
{
    Inner mut inner
    fn move := () -> i32
    {
        exit inner.p.shift(2)
    }
}

Outer old{a}
print(a.move())
print(a.inner.p.x)
